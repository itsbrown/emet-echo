2. Backend Code (Python + Flask example)
Add these routes and functions to your main app file:
Pythonfrom flask import Flask, request, jsonify
import requests
import json
import os
from datetime import datetime, timedelta

app = Flask(__name__)

BEARER_TOKEN = os.getenv("X_BEARER_TOKEN")   # Set this in your environment / secrets
HANDLES_FILE = 'x_handles.json'

# ====================== HANDLE MANAGEMENT ======================
def load_handles():
    if os.path.exists(HANDLES_FILE):
        with open(HANDLES_FILE, 'r') as f:
            return json.load(f).get('handles', [])
    return []

def save_handles(handles):
    with open(HANDLES_FILE, 'w') as f:
        json.dump({'handles': handles}, f)

@app.route('/api/save-x-handles', methods=['POST'])
def save_x_handles():
    data = request.get_json()
    handles = [h.strip() for h in data.get('handles', []) if h.strip()]
    save_handles(handles)
    return jsonify({"status": "success", "handles": handles})

@app.route('/api/get-x-handles', methods=['GET'])
def get_x_handles():
    handles = load_handles()
    return jsonify({"handles": handles})

# ====================== PULL LAST 24H POSTS ======================
def get_user_id(username):
    url = f"https://api.x.com/2/users/by/username/{username}"
    headers = {"Authorization": f"Bearer {BEARER_TOKEN}"}
    r = requests.get(url, headers=headers)
    if r.status_code == 200:
        return r.json()["data"]["id"]
    return None

def get_last_24h_tweets(username, max_results=15):
    user_id = get_user_id(username)
    if not user_id:
        return []

    start_time = (datetime.utcnow() - timedelta(hours=24)).strftime("%Y-%m-%dT%H:%M:%SZ")

    url = f"https://api.x.com/2/users/{user_id}/tweets"
    headers = {"Authorization": f"Bearer {BEARER_TOKEN}"}
    params = {
        "max_results": max_results,
        "start_time": start_time,
        "tweet.fields": "created_at,text,public_metrics"
    }

    r = requests.get(url, headers=headers, params=params)
    if r.status_code == 200:
        tweets = r.json().get("data", [])
        return [{
            "handle": f"@{username}",
            "text": t.get("text", ""),
            "created_at": t.get("created_at", ""),
            "likes": t["public_metrics"].get("like_count", 0),
            "reposts": t["public_metrics"].get("retweet_count", 0),
            "views": t["public_metrics"].get("impression_count", "N/A")
        } for t in tweets]
    return []

@app.route('/api/get-last-24h-x-posts', methods=['GET'])
def get_last_24h_x_posts():
    handles = load_handles()
    all_posts = []
    for handle in handles:
        posts = get_last_24h_tweets(handle)
        all_posts.extend(posts)
    
    # Sort by newest first
    all_posts.sort(key=lambda x: x["created_at"], reverse=True)
    return jsonify(all_posts)
