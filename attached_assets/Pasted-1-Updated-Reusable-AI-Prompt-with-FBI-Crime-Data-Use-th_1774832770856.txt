2. Updated Python Snippet – BLS + Census BFS + FBI Crime Data
This version adds a simple FBI Crime Data pull (using the public USA.gov Crime API wrapper, which is the most accessible endpoint). It returns national-level violent and property crime rates/trends where available.
Pythonimport requests
import json
from datetime import datetime

def get_bls_unemployment(start_year="2023", end_year=None):
    if end_year is None:
        end_year = datetime.now().year
    headers = {'Content-type': 'application/json'}
    data = json.dumps({
        "seriesid": ["LNS14000000"],  # National unemployment rate (seasonally adjusted)
        "startyear": start_year,
        "endyear": str(end_year)
    })
    response = requests.post('https://api.bls.gov/publicAPI/v2/timeseries/data/', data=data, headers=headers)
    if response.status_code == 200:
        json_data = response.json()
        if json_data.get('status') == 'REQUEST_SUCCEEDED':
            series = json_data['Results']['series'][0]
            latest = series['data'][0]
            return {
                "unemployment_rate": float(latest['value']),
                "period": f"{latest['year']}-{latest['period']}"
            }
    return {"error": "Failed to fetch BLS data"}
